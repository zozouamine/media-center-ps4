cmake_minimum_required(VERSION 3.20)
project(PS4MediaCenter VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MC_SANITIZERS "Instrument host validation with ASan and UBSan" OFF)
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/build_ffmpeg.py" host
    RESULT_VARIABLE MC_FFMPEG_RESULT OUTPUT_VARIABLE MC_FFMPEG_PREFIX OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT MC_FFMPEG_RESULT EQUAL 0)
    message(FATAL_ERROR "Static FFmpeg host build failed; inspect build/ffmpeg-host/build.log")
endif()
add_library(mc_ffmpeg INTERFACE)
target_include_directories(mc_ffmpeg INTERFACE "${MC_FFMPEG_PREFIX}/include")
foreach(MC_AVLIB avformat avcodec swresample swscale avutil)
    target_link_libraries(mc_ffmpeg INTERFACE "${MC_FFMPEG_PREFIX}/lib/lib${MC_AVLIB}.a")
endforeach()
target_link_libraries(mc_ffmpeg INTERFACE "${CMAKE_SOURCE_DIR}/build/dav1d-host/install/lib/libdav1d.a")
target_link_libraries(mc_ffmpeg INTERFACE "${CMAKE_SOURCE_DIR}/build/zlib-host/install/lib/libz.a")
include(vendor/libwebp/decoder_sources.cmake)
include(vendor/libtiff/decoder_sources.cmake)
include(vendor/zlib/decoder_sources.cmake)
set_source_files_properties(${MC_WEBP_SOURCES} PROPERTIES COMPILE_DEFINITIONS HAVE_CONFIG_H=1)
add_library(media_center_core
    src/library/types.cpp src/library/indexer.cpp src/input/input.cpp
    src/storage/read_only.cpp src/storage/directory.cpp src/storage/path_io.c src/ui/browser.cpp
    src/ui/canvas.cpp src/settings/store.cpp src/playlists/playlist.cpp
    src/image/image.cpp src/image/ico.cpp src/image/tiff.cpp src/audio/decoder.cpp src/audio/metadata.cpp src/audio/vorbis.c
    src/video/avi.cpp src/video/decoder.cpp src/media/engine.cpp src/platform/clock.cpp
    src/platform/host/headless_audio.cpp
    src/ui/view.cpp src/app/application.cpp
    src/platform/ps4/stat_compat.c
    src/platform/ps4/startup_trace.cpp
    src/platform/ps4/usb_access.cpp src/platform/ps4/goldhen_sdk.c src/platform/ps4/goldhen_result.c
    ${MC_WEBP_SOURCES}
    ${MC_TIFF_SOURCES} ${MC_ZLIB_SOURCES}
)
target_include_directories(media_center_core PUBLIC src PRIVATE vendor vendor/libwebp vendor/libtiff/libtiff vendor/zlib vendor/goldhen-sdk/include)
target_compile_definitions(media_center_core PRIVATE WEBP_USE_THREAD=1)
target_link_libraries(media_center_core PUBLIC Threads::Threads mc_ffmpeg)
target_compile_options(media_center_core PRIVATE -Wall -Wextra -Wpedantic)
if(MC_SANITIZERS)
    target_compile_options(media_center_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(media_center_core PUBLIC -fsanitize=address,undefined)
endif()
if(APPLE)
    enable_language(OBJCXX)
    add_library(media_center_native src/platform/macos/macos.mm)
    target_include_directories(media_center_native PUBLIC src)
    target_compile_options(media_center_native PRIVATE -fobjc-arc)
    target_link_libraries(media_center_native PUBLIC media_center_core "-framework Cocoa" "-framework QuartzCore" "-framework GameController" "-framework AudioUnit" "-framework CoreAudio")
    add_executable(ps4-media-center src/app/main.cpp)
    target_link_libraries(ps4-media-center PRIVATE media_center_native)
endif()
enable_testing()
add_executable(media-center-tests tests/test_main.cpp tests/storage_tests.cpp tests/input_tests.cpp tests/media_tests.cpp tests/extended_tests.cpp tests/video_tests.cpp)
target_include_directories(media-center-tests PRIVATE tests)
target_link_libraries(media-center-tests PRIVATE media_center_core)
add_test(NAME portable_core COMMAND media-center-tests "${CMAKE_SOURCE_DIR}")
if(APPLE)
    target_sources(media-center-tests PRIVATE src/platform/ps4/condition_compat.cpp)
    target_compile_definitions(media-center-tests PRIVATE MC_VALIDATE_PS4_RUNTIME=1)
    add_executable(ps4-runtime-tests tests/ps4_runtime_tests.cpp src/platform/ps4/condition_compat.cpp)
    target_include_directories(ps4-runtime-tests PRIVATE src)
    target_compile_definitions(ps4-runtime-tests PRIVATE MC_VALIDATE_PS4_RUNTIME=1)
    target_link_libraries(ps4-runtime-tests PRIVATE media_center_core)
    add_test(NAME ps4_runtime_compat COMMAND ps4-runtime-tests "${CMAKE_SOURCE_DIR}")
    set_tests_properties(ps4_runtime_compat PROPERTIES TIMEOUT 30)
    # Run the production PS4 source-open path with host handles. No libkernel
    # mock is linked: reintroducing a native fallback must fail this target.
    add_executable(ps4-source-probe-tests tests/ps4_source_probe_tests.cpp src/storage/read_only.cpp)
    target_include_directories(ps4-source-probe-tests PRIVATE src)
    target_compile_definitions(ps4-source-probe-tests PRIVATE MC_PS4=1 MC_PS4_SELF_PROFILE="default")
    target_link_libraries(ps4-source-probe-tests PRIVATE media_center_core)
    add_test(NAME ps4_source_probe_simulation COMMAND ps4-source-probe-tests "${CMAKE_SOURCE_DIR}")
    set_tests_properties(ps4_source_probe_simulation PROPERTIES TIMEOUT 30)
    add_library(ps4-private-path-simulation OBJECT src/storage/path_io.c src/platform/ps4/startup_trace.cpp src/settings/store.cpp)
    target_include_directories(ps4-private-path-simulation PRIVATE src tests/ps4_stubs)
    target_compile_definitions(ps4-private-path-simulation PRIVATE MC_PS4=1 open=mc_test_open openat=mc_test_openat)
    add_executable(ps4-private-io-tests tests/ps4_private_io_tests.cpp $<TARGET_OBJECTS:ps4-private-path-simulation>)
    target_include_directories(ps4-private-io-tests PRIVATE src tests/ps4_stubs)
    target_compile_definitions(ps4-private-io-tests PRIVATE MC_PS4=1)
    target_link_libraries(ps4-private-io-tests PRIVATE media_center_core)
    if(MC_SANITIZERS)
        target_compile_options(ps4-private-path-simulation PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endif()
    add_test(NAME ps4_private_io_simulation COMMAND ps4-private-io-tests "${CMAKE_SOURCE_DIR}")
    set_tests_properties(ps4_private_io_simulation PROPERTIES TIMEOUT 30)
    add_library(ps4-direct-usb-path-simulation OBJECT src/storage/path_io.c)
    target_include_directories(ps4-direct-usb-path-simulation PRIVATE src)
    target_compile_definitions(ps4-direct-usb-path-simulation PRIVATE MC_PS4=1 MC_VALIDATE_PS4_DIRECT_USB=1 open=mc_test_usb_open)
    add_library(ps4-direct-usb-simulation OBJECT src/storage/read_only.cpp src/platform/ps4/usb_access.cpp)
    target_include_directories(ps4-direct-usb-simulation PRIVATE src vendor/goldhen-sdk/include)
    target_compile_definitions(ps4-direct-usb-simulation PRIVATE MC_PS4=1 MC_PS4_SELF_PROFILE="default" MC_VALIDATE_PS4_DIRECT_USB=1)
    add_executable(ps4-direct-usb-tests tests/ps4_direct_usb_tests.cpp $<TARGET_OBJECTS:ps4-direct-usb-simulation> $<TARGET_OBJECTS:ps4-direct-usb-path-simulation>)
    target_include_directories(ps4-direct-usb-tests PRIVATE src)
    target_compile_definitions(ps4-direct-usb-tests PRIVATE MC_PS4=1)
    target_link_libraries(ps4-direct-usb-tests PRIVATE media_center_core)
    if(MC_SANITIZERS)
        target_compile_options(ps4-direct-usb-simulation PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_compile_options(ps4-direct-usb-path-simulation PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endif()
    add_test(NAME ps4_direct_usb_simulation COMMAND ps4-direct-usb-tests "${CMAKE_SOURCE_DIR}")
    set_tests_properties(ps4_direct_usb_simulation PROPERTIES TIMEOUT 30)
endif()
if(APPLE)
    add_executable(media-center-audio-probe tests/native_audio_probe.cpp)
    target_link_libraries(media-center-audio-probe PRIVATE media_center_native)
endif()
